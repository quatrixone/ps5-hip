cmake_minimum_required(VERSION 3.20)
project(ps5hip LANGUAGES CXX)
option(PS5HIP_SANITIZE "Enable host address/undefined-behavior checks" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ps5hip_object src/code_object.cpp src/launch.cpp)
target_include_directories(ps5hip_object PUBLIC include)
target_compile_options(ps5hip_object PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_CROSSCOMPILING)
  add_library(ps5hip_native_memory src/native_memory.cpp)
  target_link_libraries(ps5hip_native_memory PUBLIC ps5hip_object)
  target_compile_options(ps5hip_native_memory PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(native-memory-smoke tools/native_memory_smoke.cpp)
  target_link_libraries(native-memory-smoke PRIVATE ps5hip_native_memory)
endif()
if(PS5HIP_SANITIZE)
  target_compile_options(ps5hip_object PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ps5hip_object PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ps5hip-inspect tools/inspect.cpp)
target_link_libraries(ps5hip-inspect PRIVATE ps5hip_object)
include(CTest)
if(BUILD_TESTING)
  add_executable(object_test tests/object_test.cpp)
  target_link_libraries(object_test PRIVATE ps5hip_object)
  add_executable(resource_test tests/resource_test.cpp)
  target_link_libraries(resource_test PRIVATE ps5hip_object)
  add_executable(launch_test tests/launch_test.cpp)
  target_link_libraries(launch_test PRIVATE ps5hip_object)
  foreach(target gfx1013 gfx1030)
    add_test(NAME object_${target} COMMAND object_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/canary.hsaco ${target})
    add_test(NAME resources_${target} COMMAND resource_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/resources/resources.hsaco)
    add_test(NAME launch_${target} COMMAND launch_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/canary.hsaco)
  endforeach()
  # Only platform declarations are needed; host libc headers remain host headers.
  file(READ "${CMAKE_SOURCE_DIR}/dependencies.json" PS5HIP_HOST_LOCK)
  string(JSON PS5HIP_SDK_REV GET "${PS5HIP_HOST_LOCK}" sdk revision)
  set(PS5HIP_PLATFORM_HEADERS "${CMAKE_SOURCE_DIR}/.deps/sdk-source-${PS5HIP_SDK_REV}/platform/include")
  if(EXISTS "${PS5HIP_PLATFORM_HEADERS}/ps5platform/kernel.h")
    add_executable(native_memory_test tests/native_memory_test.cpp src/native_memory.cpp)
    target_include_directories(native_memory_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
    target_link_libraries(native_memory_test PRIVATE ps5hip_object)
    add_test(NAME native_memory_mock COMMAND native_memory_test)
  else()
    message(STATUS "Native memory mock test unavailable: prepare the pinned SDK first")
  endif()
endif()
