cmake_minimum_required(VERSION 3.20)
project(ps5hip LANGUAGES CXX)
option(PS5HIP_SANITIZE "Enable host address/undefined-behavior checks" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ps5hip_object src/code_object.cpp src/launch.cpp)
target_include_directories(ps5hip_object PUBLIC include)
target_compile_options(ps5hip_object PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_CROSSCOMPILING)
  add_library(ps5hip_native_memory src/native_memory.cpp src/submission.cpp src/agc_backend.cpp)
  target_link_libraries(ps5hip_native_memory PUBLIC ps5hip_object)
  target_compile_options(ps5hip_native_memory PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(native-memory-smoke tools/native_memory_smoke.cpp)
  target_link_libraries(native-memory-smoke PRIVATE ps5hip_native_memory)
endif()
if(PS5HIP_SANITIZE)
  target_compile_options(ps5hip_object PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ps5hip_object PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ps5hip-inspect tools/inspect.cpp)
target_link_libraries(ps5hip-inspect PRIVATE ps5hip_object)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME title_staging COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/title_stage_test.py)
  add_test(NAME canary_log_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/canary_log_test.py)
  add_test(NAME compiler_version_lock COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/compiler_version_test.py)
  add_test(NAME native_case_identity COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/native_case_test.py)
  if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/elevation/client/protocol.hpp")
    add_executable(elevation_protocol_test tests/elevation_protocol_test.cpp)
    set_property(TARGET elevation_protocol_test PROPERTY CXX_STANDARD 20)
    target_include_directories(elevation_protocol_test PRIVATE
      "${CMAKE_SOURCE_DIR}/.deps/elevation/client" "${CMAKE_SOURCE_DIR}/.deps/elevation/lapy/source")
    target_link_libraries(elevation_protocol_test PRIVATE ps5hip_object)
    add_test(NAME elevation_wire_contract COMMAND elevation_protocol_test)
  else()
    message(STATUS "Elevation wire test unavailable: prepare the pinned helper first")
  endif()
  file(READ "${CMAKE_SOURCE_DIR}/controller-inputs.json" PS5HIP_CONTROLLER_LOCK)
  file(READ "${CMAKE_SOURCE_DIR}/title.json" PS5HIP_TITLE_CONFIG)
  string(JSON PS5HIP_TITLE_ID GET "${PS5HIP_TITLE_CONFIG}" title_id)
  string(JSON PS5HIP_PROTOCOL_REV GET "${PS5HIP_CONTROLLER_LOCK}" revision)
  set(PS5HIP_PROTOCOL_DIR "${CMAKE_SOURCE_DIR}/.deps/protocol-${PS5HIP_PROTOCOL_REV}")
  if(EXISTS "${PS5HIP_PROTOCOL_DIR}/launch.c" AND EXISTS "${PS5HIP_PROTOCOL_DIR}/close.c")
    foreach(mode launch close)
      add_executable(controller_${mode}_test tests/controller_test.cpp)
      target_link_libraries(controller_${mode}_test PRIVATE ps5hip_object)
      target_compile_definitions(controller_${mode}_test PRIVATE BOOTSTRAP_TITLE_ID="${PS5HIP_TITLE_ID}" PS5HIP_CONTROLLER_SOURCE="${PS5HIP_PROTOCOL_DIR}/${mode}.c")
      if(mode STREQUAL "launch")
        target_compile_definitions(controller_${mode}_test PRIVATE PS5HIP_TEST_LAUNCH=1)
      endif()
      add_test(NAME controller_${mode}_model COMMAND controller_${mode}_test)
    endforeach()
  else()
    message(STATUS "Controller mock tests unavailable: build the pinned controllers first")
  endif()
  add_executable(native_runtime_test tests/native_runtime_test.cpp)
  target_link_libraries(native_runtime_test PRIVATE ps5hip_object)
  add_test(NAME native_runtime_oracle COMMAND native_runtime_test)
  add_executable(native_shutdown_test tests/native_shutdown_test.cpp tools/native_shutdown_probe.cpp)
  target_link_libraries(native_shutdown_test PRIVATE ps5hip_object)
  add_test(NAME native_shutdown_observer COMMAND native_shutdown_test)
  add_executable(native_shell_exit_test tests/native_shell_exit_test.cpp)
  target_link_libraries(native_shell_exit_test PRIVATE ps5hip_object)
  add_test(NAME native_shell_exit_model COMMAND native_shell_exit_test)
  add_executable(object_test tests/object_test.cpp)
  target_link_libraries(object_test PRIVATE ps5hip_object)
  add_executable(resource_test tests/resource_test.cpp)
  target_link_libraries(resource_test PRIVATE ps5hip_object)
  add_executable(launch_test tests/launch_test.cpp)
  target_link_libraries(launch_test PRIVATE ps5hip_object)
  foreach(target gfx1013 gfx1030)
    add_test(NAME object_${target} COMMAND object_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/canary.hsaco ${target})
    add_test(NAME resources_${target} COMMAND resource_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/resources/resources.hsaco)
    add_test(NAME launch_${target} COMMAND launch_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/canary.hsaco)
  endforeach()
  # Only platform declarations are needed; host libc headers remain host headers.
  file(READ "${CMAKE_SOURCE_DIR}/dependencies.json" PS5HIP_HOST_LOCK)
  string(JSON PS5HIP_SDK_REV GET "${PS5HIP_HOST_LOCK}" sdk revision)
  set(PS5HIP_PLATFORM_HEADERS "${CMAKE_SOURCE_DIR}/.deps/sdk-source-${PS5HIP_SDK_REV}/platform/include")
  if(EXISTS "${PS5HIP_PLATFORM_HEADERS}/ps5platform/kernel.h")
    add_executable(native_memory_test tests/native_memory_test.cpp src/native_memory.cpp)
    target_include_directories(native_memory_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
    target_link_libraries(native_memory_test PRIVATE ps5hip_object)
    add_test(NAME native_memory_mock COMMAND native_memory_test)
    add_executable(submission_test tests/submission_test.cpp src/submission.cpp src/native_memory.cpp)
    target_include_directories(submission_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
    target_link_libraries(submission_test PRIVATE ps5hip_object)
    add_test(NAME submission_model COMMAND submission_test)
    add_executable(agc_backend_test tests/agc_backend_test.cpp src/agc_backend.cpp)
    target_include_directories(agc_backend_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
    target_link_libraries(agc_backend_test PRIVATE ps5hip_object)
    add_test(NAME agc_boundary_model COMMAND agc_backend_test)
    add_test(NAME agc_init_failure_model COMMAND agc_backend_test init_failure)
    foreach(kernel indexed_store vector_add)
      add_executable(canary_${kernel}_test tests/native_canary_test.cpp src/native_memory.cpp src/submission.cpp src/agc_backend.cpp)
      target_include_directories(canary_${kernel}_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
      target_compile_definitions(canary_${kernel}_test PRIVATE PS5HIP_CANARY_TARGET="gfx1013")
      if(kernel STREQUAL "vector_add")
        target_compile_definitions(canary_${kernel}_test PRIVATE PS5HIP_CANARY_VECTOR_ADD=1)
      endif()
      target_link_libraries(canary_${kernel}_test PRIVATE ps5hip_object)
      add_test(NAME canary_${kernel}_oracle COMMAND canary_${kernel}_test)
    endforeach()
    add_executable(arithmetic_suite_test tests/arithmetic_suite_test.cpp src/submission.cpp src/agc_backend.cpp)
    target_include_directories(arithmetic_suite_test PRIVATE "${PS5HIP_PLATFORM_HEADERS}")
    target_compile_definitions(arithmetic_suite_test PRIVATE PS5HIP_CANARY_TARGET="gfx1013" PS5HIP_CANARY_ARITHMETIC_SUITE=1)
    target_link_libraries(arithmetic_suite_test PRIVATE ps5hip_object)
    foreach(target gfx1013 gfx1030)
      add_test(NAME arithmetic_suite_${target}_model COMMAND arithmetic_suite_test ${CMAKE_SOURCE_DIR}/build/canaries/${target}/canary.hsaco)
    endforeach()
  else()
    message(STATUS "Native boundary mock tests unavailable: prepare the pinned SDK first")
  endif()
endif()
